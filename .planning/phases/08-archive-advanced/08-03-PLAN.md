def create_archive_test_data():
    """Crea 10 ordini SPEDITO con tempi realistici per test archivio"""
    from .database import OrderManager
    from .models import Order, ProcessingStep, OrderStatus, get_session
    from datetime import datetime, timedelta
    import uuid

    session = get_session()
    try:
        # Controlla se test data già esiste
        existing = session.query(Order).filter(Order.status == OrderStatus.SPEDITO.value).count()
        if existing > 0:
            print(f"[INFO] Test data archivio già presente ({existing} ordini SPEDITO)")
            return

        # Crea 10 ordini con clienti e articoli variati
        test_clients = [
            ('ABC Spa', ['Staffa A', 'Bullone B', 'Piastra C']),
            ('XYZ Srl', ['Telaio 1', 'Gancio 2']),
            ('DEF Industries', ['Supporto Alpha', 'Braccetto Beta', 'Guida Gamma']),
            ('GHI Metallica', ['Tubo 1', 'Lastra 2']),
            ('JKL Metal', ['Angolare A', 'Profilo B', 'Barra C', 'Lamiera D']),
        ]

        base_date = datetime.utcnow() - timedelta(days=30)

        for i in range(10):
            cliente_idx = i % len(test_clients)
            cliente, article_names = test_clients[cliente_idx]

            articles = [
                {
                    'name': name,
                    'code': f'ART-{i:03d}-{j:02d}',
                    'qty': (j + 1) * 10,
                    'required_phases': ['LASER', 'PIEGA', 'SALDATURA'] if j % 2 == 0 else ['LASER', 'PIEGA']
                }
                for j, name in enumerate(article_names)
            ]

            data_consegna = (base_date + timedelta(days=i+1)).isoformat()

            # Crea ordine
            order = OrderManager.create_order(
                cliente=f'{cliente} - Ordine {i+1}',
                data_consegna=data_consegna,
                articles=articles
            )

            # Simula completamento fasi con tempi realistici
            # LASER: 2-3 ore, PIEGA: 1-2 ore, SALDATURA: 2-4 ore
            phase_durations = {
                'LASER': (2 + i % 3) * 60 + (i * 15),
                'PIEGA': (1 + i % 2) * 60 + (i * 10),
                'SALDATURA': (2 + i % 3) * 60 + (i * 20),
            }

            # Carica ordine
            session_reload = get_session()
            order_reload = session_reload.query(Order).filter(Order.id == order.id).first()

            current_time = base_date + timedelta(days=i+1, hours=8)  # Inizio lavoro alle 8 AM

            for step in order_reload.processing_steps:
                phase = step.fase
                if phase in phase_durations:
                    step.timestamp_inizio = current_time
                    duration_minutes = phase_durations[phase]
                    step.timestamp_fine = current_time + timedelta(minutes=duration_minutes)
                    step.operatore = f'Operaio {phase[0]}'

                    # Marca completed_articles
                    if order_reload.articles:
                        step.completed_articles = list(range(len(order_reload.articles)))

                    current_time = step.timestamp_fine + timedelta(minutes=15)  # 15 min gap tra fasi

            order_reload.status = OrderStatus.SPEDITO.value
            session_reload.commit()
            session_reload.close()

        print(f"[OK] Creati 10 ordini SPEDITO per test archivio")

    except Exception as e:
        session.rollback()
        print(f"[WARN] Errore creazione test data archivio: {e}")
    finally:
        session.close()

class TestArchiveExport:
    """Test export CSV"""

    async def test_csv_export_downloads(self, page, context):
        """Bottone CSV export scarica file"""
        # Ascolta download
        download_promise = asyncio.create_task(context.expect_event('page.download'))

        await page.goto('http://localhost:5000/admin.html')
        await page.select_option('#user-select', 'marco-admin')
        await page.click('button:text("Accedi")')
        await page.click('.archive-btn')
        await page.wait_for_selector('#archive-export-csv', timeout=5000)

        await page.click('#archive-export-csv')

        try:
            download = await asyncio.wait_for(download_promise, timeout=10)
            filename = download.suggested_filename

            # Verifica che filename è CSV
            assert filename.endswith('.csv'), f"File deve essere CSV, got {filename}"

            # Salva file temporaneo
            temp_path = Path(f'/tmp/{filename}')
            await download.save_as(temp_path)

            # Verifica che file contiene dati
            with open(temp_path, 'r') as f:
                csv_reader = csv.reader(f)
                rows = list(csv_reader)

            assert len(rows) >= 1, "CSV deve avere almeno header"
            # Header check
            header = rows[0]
            assert 'numero_ordine' in header or 'cliente' in header, "CSV deve avere colonne ordine/cliente"

            # Cleanup
            temp_path.unlink()

        except asyncio.TimeoutError:
            pytest.skip("Download timeout - potrebbe essere timing issue")
