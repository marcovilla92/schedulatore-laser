2. **Funzione principale loadArchiveOrders()**:
   ```javascript
   async function loadArchiveOrders() {
     try {
       // Costruisci URL con query params da archiveState
       const params = new URLSearchParams();
       params.append('page', archiveState.currentPage);
       params.append('page_size', archiveState.pageSize);
       params.append('sort_by', archiveState.sortBy);
       params.append('sort_order', archiveState.sortOrder);

       if (archiveState.filters.date_from) {
         params.append('date_from', archiveState.filters.date_from);
       }
       if (archiveState.filters.date_to) {
         params.append('date_to', archiveState.filters.date_to);
       }
       if (archiveState.filters.cliente) {
         params.append('cliente', archiveState.filters.cliente);
       }
       if (archiveState.filters.phase) {
         params.append('phase', archiveState.filters.phase);
       }

       showElement('archive-loading');
       hideElement('archive-table-container');
       hideElement('archive-empty');

       const response = await fetch(`/api/archive/orders?${params}`);
       if (!response.ok) {
         const err = await response.json();
         console.error('API error:', err);
         showElement('archive-empty');
         hideElement('archive-loading');
         return;
       }

       const data = await response.json();
       hideElement('archive-loading');

       if (data.orders.length === 0) {
         showElement('archive-empty');
         hideElement('archive-table-container');
       } else {
         renderArchiveTable(data.orders);
         showElement('archive-table-container');
         hideElement('archive-empty');
       }

       // Aggiorna pagination info
       archiveState.totalPages = Math.ceil(data.total / archiveState.pageSize);
       updatePaginationInfo();

     } catch (error) {
       console.error('Error loading archive:', error);
       showElement('archive-empty');
       hideElement('archive-loading');
     }
   }
   ```

3. **Funzione renderArchiveTable(orders)**:

   function renderArchiveTable(orders) {
     const tbody = document.getElementById('archive-tbody');
     tbody.innerHTML = '';

     orders.forEach(order => {
       const row = document.createElement('tr');

       // Formatta date
       const completionDate = new Date(order.data_completamento).toLocaleDateString('it-IT');

       // Estrai tempi per fase (o '--' se non presente)
       const laserTime = order.tempi_per_fase.LASER || '--';
       const piegaTime = order.tempi_per_fase.PIEGA || '--';
       const saldaruraTime = order.tempi_per_fase.SALDATURA || '--';

       row.innerHTML = `
         <td><strong>${order.numero_ordine || order.id.substring(0, 8)}</strong></td>
         <td>${order.cliente}</td>
         <td>${completionDate}</td>
         <td class="time-cell">${laserTime}</td>
         <td class="time-cell">${piegaTime}</td>
         <td class="time-cell">${saldaruraTime}</td>
         <td><strong>${order.tempo_totale}</strong></td>
         <td>${order.numero_articoli}</td>
         <td>
           <button class="btn-small" onclick="showArchiveDetails('${order.id}')" title="Visualizza dettagli">
             Dettagli
           </button>
         </td>
       `;
       tbody.appendChild(row);
     });
   }

   ```javascript
   async function showArchiveDetails(orderId) {
     try {
       const response = await fetch(`/api/archive/orders/${orderId}/details`);
       if (!response.ok) {
         const err = await response.json();
         alert(`Errore: ${err.error}`);
         return;
       }

       const order = await response.json();

       // Costruisci HTML modale
       const modal = document.getElementById('archive-modal');
       const modalBody = document.getElementById('archive-modal-body');

       // Formatta date
       const creationDate = new Date(order.data_creazione).toLocaleDateString('it-IT', {
         year: 'numeric',
         month: 'long',
         day: 'numeric',
         hour: '2-digit',
         minute: '2-digit'
       });

       const deliveryDate = new Date(order.data_consegna).toLocaleDateString('it-IT', {
         year: 'numeric',
         month: 'long',
         day: 'numeric'
       });

       // Costruisci HTML per tabella tempi per fase
       let phaseTimesHTML = '';
       for (const [phase, times] of Object.entries(order.tempi_per_fase)) {
         if (times) {
           const start = new Date(times.inizio).toLocaleTimeString('it-IT');
           const end = new Date(times.fine).toLocaleTimeString('it-IT');
           phaseTimesHTML += `
             <tr>
               <td>${phase}</td>
               <td>${start}</td>
               <td>${end}</td>
               <td><strong>${times.durata}</strong></td>
             </tr>
           `;
         }
       }

       // Costruisci HTML per articoli
       let articlesHTML = '';
       if (order.articoli && order.articoli.length > 0) {
         order.articoli.forEach((art, idx) => {
           const phases = art.required_phases ? art.required_phases.join(', ') : '--';
           articlesHTML += `
             <tr>
               <td>${art.name || `Articolo ${idx + 1}`}</td>
               <td>${art.code || '--'}</td>
               <td>${art.qty || 0}</td>
               <td>${phases}</td>
             </tr>
           `;
         });
       }

       modalBody.innerHTML = `
         <h2>Dettagli Ordine</h2>
         <div style="margin-top: 16px;">
           <div style="display: grid; grid-template-columns: repeat(2, 1fr); gap: 12px; margin-bottom: 20px;">
             <div>
               <strong>Numero Ordine:</strong> ${order.numero_ordine || order.id.substring(0, 8)}
             </div>
             <div>
               <strong>Cliente:</strong> ${order.cliente}
             </div>
             <div>
               <strong>Data Creazione:</strong> ${creationDate}
             </div>
             <div>
               <strong>Data Consegna:</strong> ${deliveryDate}
             </div>
             <div>
               <strong>Numero Articoli:</strong> ${order.numero_articoli}
             </div>
             <div>
               <strong>Tempo Totale:</strong> ${order.tempo_totale}
             </div>
             <div>
               <strong>Status:</strong> ${order.status}
             </div>
           </div>

           <h3 style="margin-top: 20px; margin-bottom: 12px;">Tempi per Fase</h3>
           <table class="data-table" style="width: 100%;">
             <thead>
               <tr>
                 <th>Fase</th>
                 <th>Inizio</th>
                 <th>Fine</th>
                 <th>Durata</th>
               </tr>
             </thead>
             <tbody>
               ${phaseTimesHTML}
             </tbody>
           </table>

           <h3 style="margin-top: 20px; margin-bottom: 12px;">Articoli</h3>

   async function exportArchiveCSV() {
     try {
       // Costruisci URL con gli stessi filtri applicati
       const params = new URLSearchParams();
       if (archiveState.filters.date_from) {
         params.append('date_from', archiveState.filters.date_from);
       }
       if (archiveState.filters.date_to) {
         params.append('date_to', archiveState.filters.date_to);
       }
       if (archiveState.filters.cliente) {
         params.append('cliente', archiveState.filters.cliente);
       }
       if (archiveState.filters.phase) {
         params.append('phase', archiveState.filters.phase);
       }

       const response = await fetch(`/api/archive/export/csv?${params}`);
       if (!response.ok) {
         const err = await response.json();
         alert(`Errore: ${err.error}`);
         return;
       }

       // Estrai filename dal header Content-Disposition o usa default
       const contentDisposition = response.headers.get('Content-Disposition');
       let filename = 'archive_export.csv';
       if (contentDisposition) {
         const match = contentDisposition.match(/filename="(.+)"/);
         if (match) filename = match[1];
       }

       // Crea blob e scarica
       const blob = await response.blob();
       const url = window.URL.createObjectURL(blob);
       const a = document.createElement('a');
       a.href = url;
       a.download = filename;
       document.body.appendChild(a);
       a.click();
       window.URL.revokeObjectURL(url);
       document.body.removeChild(a);

     } catch (error) {
       console.error('Error exporting CSV:', error);
       alert('Errore durante il download del CSV');
     }
   }
   ```
