**2. loadOrders() con fetch/render separation (FILT-04):**
```javascript
async function loadOrders(forceRender) {
  try {
    const response = await fetch(`${API_URL}/orders`);
    const orders = await response.json();
    const orderList = Array.isArray(orders) ? orders : [];

    const { changed, newHash } = hasDataChanged(orderList, lastDataHash);
    if (!forceRender && !changed) return;
    lastDataHash = newHash;

    allOrders = orderList;
    lastRefreshTime = Date.now();
    applyFiltersAndRender();
  } catch (error) {
    showMessage('Errore connessione al server', 'error');
  }
}
```

**4. KPI computation (DASH-02):**
```javascript
function updateKPIs(activeOrders) {
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const weekEnd = new Date(today);
  weekEnd.setDate(weekEnd.getDate() + 7);

  const ordiniAttivi = activeOrders.length;
  const scadenzeOggi = activeOrders.filter(o => {
    if (!o.data_consegna) return false;
    const d = new Date(o.data_consegna); d.setHours(0, 0, 0, 0);
    return d.getTime() === today.getTime();
  }).length;
  const scadenzeSettimana = activeOrders.filter(o => {
    if (!o.data_consegna) return false;
    const d = new Date(o.data_consegna); d.setHours(0, 0, 0, 0);
    return d >= today && d <= weekEnd;
  }).length;
  const fasiInCorso = activeOrders.filter(o =>
    (o.processing_steps || []).some(s => s.timestamp_inizio && !s.timestamp_fine)
  ).length;

  document.getElementById('kpiAttivi').textContent = ordiniAttivi;
  document.getElementById('kpiOggi').textContent = scadenzeOggi;
  document.getElementById('kpiSettimana').textContent = scadenzeSettimana;
  document.getElementById('kpiFasiInCorso').textContent = fasiInCorso;
}
```

**5. Urgency class computation (DASH-03):**
```javascript
function getUrgencyClass(dataConsegna) {
  if (!dataConsegna) return 'urgency-futuro';
  const today = new Date(); today.setHours(0, 0, 0, 0);
  const delivery = new Date(dataConsegna); delivery.setHours(0, 0, 0, 0);
  const diff = delivery - today;
  if (diff < 0) return 'urgency-scaduto';
  if (diff === 0) return 'urgency-oggi';
  return 'urgency-futuro';
}

function getUrgencyLabel(urgencyClass) {
  if (urgencyClass === 'urgency-scaduto') return 'Scaduto';
  if (urgencyClass === 'urgency-oggi') return 'Oggi';
  return '';
}
```

**6. renderOrders() — urgency cards (DASH-03 + DASH-05):**
```javascript
function renderOrders(orders) {
  const grid = document.getElementById('ordersGrid');
  const empty = document.getElementById('emptyState');

  if (!orders.length) {
    grid.innerHTML = '';
    empty.style.display = 'block';
    return;
  }
  empty.style.display = 'none';

  // Sort by urgency: scaduto first, oggi second, futuro last
  const sorted = [...orders].sort((a, b) => {
    const urgencyOrder = { 'urgency-scaduto': 0, 'urgency-oggi': 1, 'urgency-futuro': 2 };
    return (urgencyOrder[getUrgencyClass(a.data_consegna)] || 2) - (urgencyOrder[getUrgencyClass(b.data_consegna)] || 2);
  });

  grid.innerHTML = sorted.map(order => {
    const urgency = getUrgencyClass(order.data_consegna);
    const label = getUrgencyLabel(urgency);
    const badgeClass = urgency.replace('urgency-', '');
    const articles = order.articles || [];
    const totalQty = articles.reduce((sum, a) => sum + (a.quantita || a.quantity || 0), 0);

    // Phase tags from processing_steps
    const phaseTags = (order.processing_steps || []).reduce((acc, step) => {
      const phaseName = step.phase_name || step.phase;
      if (!acc.find(p => p.name === phaseName)) {
        const isActive = step.timestamp_inizio && !step.timestamp_fine;
        const isCompleted = !!step.timestamp_fine;
        acc.push({ name: phaseName, active: isActive, completed: isCompleted });
      }
      return acc;
    }, []);

    return `
      <div class="order-card ${urgency}" onclick="openOrderModal('${order.id}')">
        <div class="order-cliente">${order.cliente || 'N/A'}</div>
        <div class="order-meta">
          <span>${order.id ? order.id.substring(0, 8) : ''}</span>
          <span>${articles.length} articoli (${totalQty} pz)</span>
          ${order.data_consegna ? `<span>${new Date(order.data_consegna).toLocaleDateString('it-IT')}</span>` : ''}
          ${label ? `<span class="urgency-badge ${badgeClass}">${label}</span>` : ''}
        </div>
        <div class="order-phases">
          ${phaseTags.map(p => `<span class="phase-tag ${p.completed ? 'completed' : p.active ? 'active' : ''}">${p.name}</span>`).join('')}
        </div>
      </div>
    `;
  }).join('');
}
```

**7. Phases summary (DASH-05):**
```javascript
function renderPhasesSummary(activeOrders) {
  const phases = ['LASER', 'PIEGA', 'SALDATURA', 'PULIZIA', 'SPEDIZIONE'];
  const container = document.getElementById('phasesSummary');

  const counts = phases.map(phase => {
    const inPhase = activeOrders.filter(o =>
      (o.processing_steps || []).some(s =>
        (s.phase_name || s.phase) === phase && s.timestamp_inizio && !s.timestamp_fine
      )
    ).length;
    return { phase, count: inPhase };
  }).filter(p => p.count > 0);

  if (!counts.length) { container.innerHTML = ''; return; }

  container.innerHTML = counts.map(p => `
    <div class="phase-summary-card">
      <div class="phase-summary-count">${p.count}</div>
      <div class="phase-summary-label">${p.phase} in corso</div>
    </div>
  `).join('');
}
```
