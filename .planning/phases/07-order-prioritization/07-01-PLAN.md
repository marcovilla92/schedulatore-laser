function sortOrdersByEDD(orders) {
  // Input: Array of order objects (from /api/phase/LASER/orders)
  // Output: Sorted array (earliest delivery date first)

  if (!Array.isArray(orders) || orders.length === 0) return orders;

  return orders.slice().sort((a, b) => {
    // Parse delivery dates safely
    const dateA = new Date(a.data_consegna);
    const dateB = new Date(b.data_consegna);

    // Handle invalid dates (fallback to 0 = no sort change)
    if (isNaN(dateA.getTime())) return 0;
    if (isNaN(dateB.getTime())) return 0;

    // EDD rule: compare dates (ascending = earliest first)
    return dateA.getTime() - dateB.getTime();
  });
}

Key implementation details:
- Use slice() to create a new array (immutable — don't modify original)
- Use getTime() for numeric date comparison (handles timezone differences)
- Handle invalid/missing dates gracefully (return 0 = no sort change, don't throw)
- Return ascending (earliest/most urgent first)

This function will be called BEFORE renderOrders() in the renderOrders flow.
