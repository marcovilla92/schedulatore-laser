📊 Algoritmo EDD (Earliest Due Date)

// 1. Carica ordini da API
orders = fetch('/api/phase/LASER/orders')

// 2. Calcola giorni rimasti per ogni ordine

  order.daysRemaining = calculateDaysRemaining(order.data_consegna)
  order.urgencyTier = getUrgencyTier(order.daysRemaining)
})

// 3. Ordina per data di scadenza (EDD)
orders.sort((a, b) => new Date(a.data_consegna) - new Date(b.data_consegna))
